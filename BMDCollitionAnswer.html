Аналіз системи колізій BMD файлу

1. Загальна архітектура системи колізій

Движок гри Angelica2 використовує дві основні структури для обробки колізій:

1.1. ESP (Extended Spatial Partitioning)

  • Просторове розбиття на візуальні листки (leaves)
  • Використовує 3DDDA алгоритм для трасування променів
  • Ідеально для статичних сцен

1.2. KD-Tree (K-Dimensional Tree)

  • Бінарне дерево розбиття простору
  • Рекурсивне трасування по дереву
  • Більш ефективний для складних сцен

2. Структура файлу колізії

Файли: .esp (ESP) та .kdt (KD-Tree)

2.1. Структура ESP файлу

struct ESPFILEHEADER { DWORD dwIdentify; // Ідентифікатор: 'AESP' (0x41455350) DWORD dwVersion; // Версія: 5 int iNumLump; // Кількість блоків даних BYTE aReserved[60]; // Зарезервовано }; enum ESP_LUMPS { EFLUMP_LEAVES = 0, // Листки (просторові комірки) EFLUMP_PLANES, // Плоскості EFLUMP_VERTICES, // Вершини EFLUMP_LEAFSIDES, // Посилання на сторони в листках EFLUMP_SIDES, // Сторони (трикутники/чотирикутники) EFLUMP_WORLDINFO, // Інформація про світ EFLUMP_BRUSHES, // Кисті (об'єми) EFLUMP_BRUSHPLANES, // Плоскості кистей NUM_ESPLUMP };

2.2. Структура KD-Tree файлу

struct KDTFILEHEADER { DWORD dwIdentify; // Ідентифікатор: 'AKDT' (0x414B4454) DWORD dwVersion; // Версія: 2 int iNumLump; // Кількість блоків даних BYTE aReserved[60]; // Зарезервовано }; enum KDT_LUMPS { KDTLUMP_VERTICES = 0, // Вершини KDTLUMP_PLANES, // Плоскості KDTLUMP_NODES, // Вузли дерева KDTLUMP_LEAFSIDES, // Посилання на сторони KDTLUMP_SIDES, // Сторони KDTLUMP_BRUSHES, // Кисті KDTLUMP_BRUSHPLANES, // Плоскості кистей NUM_KDTLUMP };

3. Основні структури даних

3.1. Плоскість (Plane)

struct EFPLANE { FLOAT vNormal[3]; // Нормаль плоскості (x, y, z) FLOAT fDist; // Відстань від початку координат }; // Формула плоскості: n·p = d // де n - нормаль, p - точка на плоскості, d - відстань

3.2. Сторона (Side)

struct EFSIDE { LONG lPlane; // Індекс плоскості LONG lFirstVert; // Перша вершина в масиві вершин LONG lNumVert; // Кількість вершин DWORD dwMeshProps; // Властивості сітки int iMtlType; // Тип матеріалу LONG lBrush; // Індекс кисті }; // Кожна сторона - це полігон (трикутник або чотирикутник) // який є елементом колізії

3.3. Листок (Leaf) - ESP

struct EFLEAF { LONG lFirstLeafSide; // Перше посилання на сторону LONG lNumLeafSides; // Кількість сторін у листку }; // Листок - це просторова комірка (воксель) // містить список сторін, що знаходяться в ній

3.4. Вузол (Node) - KD-Tree

struct KDTFNODE { int iPlane; // Індекс розділювальної площини (-1 = листок) int iParent; // Батьківський вузол int aChildren[2]; // Дочірні вузли [0] - ліворуч, [1] - праворуч int iFirstSide; // Перша сторона в списку int iNumSide; // Кількість сторін float vMins[3]; // AABB мінімуми float vMaxs[3]; // AABB максимуми }; // Вузол дерева містить розділювальну площину // яка діить простір на дві підмножини

4. Процес створення колізії

4.1. Завантаження сцени

// З Side.cpp void _CreateSides() { // 1. Завантажуємо всі меші зі сцени for (кожен меш в сцені) { // 2. Перевіряємо властивості меша if (меш.мультіфрейм || меш.pickthrough) continue; // 3. Створюємо сторони з трикутників _CreateSidesFromTriangles( кількість_вершин, кількість_індексів, вершини, індекси, властивості_меша ); } }

4.2. Об'єднання трикутників

// З Side.cpp bool _CreateSidesFromTriangles(...) { if (g_Configs.bMergeSide && iNumIndex >= 6) { // Спроба об'єднати два трикутники в чотирикутник // Якщо вони: // - лежать в одній площині // - мають спільний ребро // - утворюють опуклий чотирикутник // Об'єднання зменшує кількість полігонів // покращуючи продуктивність } }

4.3. Створення ESP світу

// З ESP_Leaf.cpp bool _CreateESPWorld() { // 1. Обчислюємо AABB всієї сцени ClearBound(g_ESPWorld.vMins, g_ESPWorld.vMaxs); for (кожна_сторона) { AddPointToBound(g_ESPWorld.vMins, g_ESPWorld.vMaxs, вершина); } // 2. Розширюємо границі для запобігання помилок int iDefLeafSize = g_Configs.iESPLeafSize; // За замовчуванням 20 float fBorderErr = (float)iDefLeafSize; // 3. Вирівнюємо границі по розміру листка for (int i = 0; i < 3; i++) { g_ESPWorld.vAlignMins[i] = floor((vMins[i] - fBorderErr) / iDefLeafSize) * iDefLeafSize; g_ESPWorld.vAlignMaxs[i] = ceil((vMaxs[i] + fBorderErr) / iDefLeafSize) * iDefLeafSize; g_ESPWorld.vLength[i] = iDefLeafSize; g_ESPWorld.vSize[i] = (int)((vAlignMaxs[i] - vAlignMins[i]) / iDefLeafSize + 0.5f); } // 4. Виділяємо пам'ять для листків int iNumLeaf = vSize[0] * vSize[1] * vSize[2]; g_ESPWorld.aLeaves = new s_ESPLEAF[iNumLeaf]; // 5. Ініціалізуємо листки _InitializeLeaves(); }

4.4. Фільтрація сторін в листки

// З ESP_Leaf.cpp void _FilterSidesIntoLeaves() { for (кожна_сторона) { // Обчислюємо AABB сторони for (кожна_вершина_сторони) { AddPointToBound(vMins, vMaxs, вершина); } // Знаходимо діапазон листків, що перетинаються з AABB for (x = min_x до max_x) { for (y = min_y до max_y) { for (z = min_z до max_z) { // Додаємо посилання на сторону в листок int iLeaf = x + y * vSize[0] + z * vSize[0] * vSize[1]; AddSideRefToLeaf(iLeaf, iSide); } } } } }

5. Алгоритми трасування

5.1. Ray Tracing (Трасування променя)

// З A3DESP.cpp bool A3DESP::RayTrace(RAYTRACERT* pTrace, const A3DVECTOR3& vStart, const A3DVECTOR3& vVelocity, FLOAT fTime) { // 1. Ініціалізація даних трасування m_Ray.vStart = vStart; m_Ray.vEnd = vStart + vVelocity * fTime; m_Ray.vDir = m_Ray.vEnd - m_Ray.vStart; m_Ray.fFraction = 1.0f; // 2. Перевірка чи промінь аксіальний if (vDir.x == 0.0f && vDir.z == 0.0f) m_Ray.iAxial = 1; // Y-аксіальний else if (vDir.y == 0.0f && vDir.z == 0.0f) m_Ray.iAxial = 0; // X-аксіальний // ... інші випадки // 3. Трасування в світ return RayToWorld(); }

5.2. 3DDDA Алгоритм (ESP)

// З A3DESP.cpp void A3DESP::Init3DDDA(A3DVECTOR3& v0, A3DVECTOR3& v1) { // Ініціалізація 3D Digital Differential Analyzer // Визначення початкового та кінцевого вокселів for (int i = 0; i < 3; i++) { vStart[i] = (int)((v0.m[i] - vAlignMins.m[i]) * vInvLength.m[i]); vEnd[i] = (int)((v1.m[i] - vAlignMins.m[i]) * vInvLength.m[i]); // Обчислення кроків та помилок if (vDelta.m[i] < 0.0f) { s[i] = -1; vSteps[i] = vStart[i] - vEnd[i]; d[i] = v0.m[i] - (vAlignMins.m[i] + vStart[i] * vLength[i]); } else { s[i] = 1; vSteps[i] = vEnd[i] - vStart[i]; d[i] = vAlignMins.m[i] + (vStart[i] + 1) * vLength[i] - v0.m[i]; } } // Визначення головної осі (з найбільшим кроком) if (vAbs.x > vAbs.y && vAbs.x > vAbs.z) iMajor = 0; // X головна // ... інші випадки }

5.3. KD-Tree Recursive Tracing

// З A3DKDTree.cpp void A3DKDTree::RecursiveTraceCheck(NODE* pNode, float f1, float f2, const A3DVECTOR3& vP1, const A3DVECTOR3& vP2) { // 1. Якщо вже знайшли ближче зіткнення - повертаємось if (m_Trace.fHitFrac <= f1) return; // 2. Якщо це листок (немає площини розбиття) if (!pNode->pPlane) { switch (m_Trace.iTrace) { case TRACE_RAY: TraceRayInLeaf(pNode); break; case TRACE_AABB: TraceAABBInLeaf(pNode); break; case TRACE_SPHERE: TraceSphereInLeaf(pNode); break; } return; } // 3. Знаходимо відстані до розділювальної площини float t1 = vP1.m[pPlane->byType] - pPlane->fDist; float t2 = vP2.m[pPlane->byType] - pPlane->fDist; // 4. Рекурсивно перевіряємо дочірні вузли if (t1 >= fOffset && t2 >= fOffset) { RecursiveTraceCheck(pNode->aChildren[0], f1, f2, vP1, vP2); } else if (t1 < -fOffset && t2 < -fOffset) { RecursiveTraceCheck(pNode->aChildren[1], f1, f2, vP1, vP2); } else { // Промінь перетинає площину - перевіряємо обидва боки float fFrac1, fFrac2; // Обчислення коефіцієнтів перетину RecursiveTraceCheck(pNode->aChildren[iSide], f1, fMid, vP1, vMid); RecursiveTraceCheck(pNode->aChildren[iSide^1], fMid, f2, vMid, vP2); } }

6. Типи колізій

6.1. Ray-Side Collision

// З A3DKDTree.cpp bool A3DKDTree::RayToSide(SIDE* pSide) { // 1. Перевірка AABB сторони з променем if (!RayToAABB(pSide->vMins, pSide->vMaxs)) return false; // 2. Тестування перетину з трикутником for (int i = 0; i < pSide->iNumVert - 2; i++) { if (CLS_RayToTriangle( m_Trace.vStart, m_Trace.vDelta, pSide->aVerts[0], pSide->aVerts[i+1], pSide->aVerts[i+2], vPoint, pSide->MeshProps.Get2SidesFlag(), &fFraction)) { // 3. Оновлення результату, якщо знайшли ближче зіткнення if (fFraction < m_Trace.fHitFrac) { m_Trace.fHitFrac = fFraction; m_Trace.fFraction = fFraction; m_Trace.pSide = pSide; m_Trace.vNormal = pSide->pPlane->vNormal; m_Trace.vPoint = vPoint; } } } return true; }

7. Оптимізації

7.1. Просторове розбиття

  • ESP: Рівномірне розбиття на вокселі
  • KD-Tree: Адаптивне розбиття по площинах

7.2. Кешування

// Використання dwTraceCnt для уникнення повторної перевірки struct ESPSIDE { DWORD dwTraceCnt; // Лічильник трасування }; // Перевірка: if (pSide->dwTraceCnt == m_dwTraceCnt) continue; // Вже перевірено в цьому кадрі

7.3. Аксіальне трасування

  • Для променів, паралельних осям
  • Спрощений алгоритм 3DDDA
  • Швидше на ~30%

struct A3DMESHPROP { DWORD dwProps; // Прапорці властивостей: // Bit 0: TwoSides - двостороння колізія // Bit 1: NoMark - не залишати слідів // Bit 2: PickThrough - пропускати при виборі // Bit 3: NoCollide - вимкнена колізія bool Get2SidesFlag() { return dwProps & 0x01; } bool GetNoMarkFlag() { return dwProps & 0x02; } bool GetPickThroughFlag() { return dwProps & 0x04; } }; enum A3DMATERIALTYPE { MAT_NONE = 0, MAT_METAL, // Метал - дзвінкий звук MAT_WOOD, // Дерево - глухий звук MAT_STONE, // Камінь - важкий звук MAT_WATER, // Вода - звук крапель MAT_EARTH, // Земля - м'який звук // ... інші типи };

10. Процес імпорту в Blender

Примітка: Базуючись на аналізі вихідного коду, процес імпорту колізії в Blender включає:

10.1. Читання файлу

  1. Зчитування заголовка файлу
  2. Визначення типу (ESP або KD-Tree)
  3. Читання блоків даних (lumps)

10.2. Конвертація даних

  1. Плоскості → Mesh faces
  2. Вершини → Mesh vertices
  3. Сторони → Mesh polygons
  4. Кисті → Boolean об'єкти або меші

10.3. Створення об'єктів

  1. Основна колізія (ліва кнопка миші)
  2. Детальна колізія (кисті)
  3. Маркери (точки для подій)

11. Висновки

Система колізій BMD файлу реалізована на основі:

  • ESP - для простих сцен, швидке трасування променів
  • KD-Tree - для складних сцен, адаптивне розбиття
  • Brush - для точної колізії складних об'єктів

Основні переваги:

  • Висока продуктивність завдяки просторовому розбиттю
  • Підтримка різних типів колізій (промінь, AABB, сфера)
  • Гнучка система властивостей матеріалів
  • Оптимізація для ігрових сцен

Аналіз файлу CollisionFile.bmd

Важливо: Файл CollisionFile.bmd насправді є MOX файлом (MOX Binary), а не ESP або KD-Tree файлом.

Структура файлу

// Заголовок файлу (зчитано з бінарного аналізу) Offset 0x00: "MOXB" - Магічний номер MOX Binary (0x4D4F5842) Offset 0x04: Версія файлу (0x00000002 = версія 2) Offset 0x08: Розмір файлу (0x1000001F = ~268 MB) // Структура заголовку MOX файлу: struct MOXFILEHEADER { DWORD dwIdentify; // "MOXB" = 0x4D4F5842 DWORD dwVersion; // Версія формату DWORD dwFileSize; // Розмір файлу };

Як реалізована колізія

На основі аналізу файлу та вихідного коду движка, колізія в CollisionFile.bmd реалізована наступним чином:

  1. Кожен меш (Box01, Box367, Object01) містить:
    • Масив вершин (A3DVERTEX)
    • Масив індексів (WORD indices)
    • Властивості меша (A3DMESHPROP)
    • Текстуру
  2. Властивості колізії зберігаються в A3DMESHPROP:
    // Кожен меш має властивості, що визначають поведінку колізії struct A3DMESHPROP { DWORD dwProps; // Прапорці властивостей A3DMATERIALTYPE MtlType; // Тип матеріалу для звуку/ефектів }; // Прапорці властивостей: FG_2SIDES = 0x0001 // Двостороння колізія FG_SHOOTTHROUGH = 0x0002 // Пропускає кулі FG_NOMARK = 0x0004 // Не залишає слідів FG_PICKTHROUGH = 0x0008 // Пропускає при виборі FG_ONGROUND = 0x0010 // Знаходиться на землі FG_BSPPVS = 0x0020 // Використовується для PVS FG_NOTRENDER = 0x0040 // Не рендериться FG_WALKIGNORE = 0x0080 // Ігнорується при ходьбі FG_NOLYFORWALK = 0x0100 // Тільки для ходьби
  3. Типи матеріалів для колізії:
    enum A3DMATERIALTYPE { A3DMTL_CONCRETE = 0, // Бетон A3DMTL_WOOD, // Дерево A3DMTL_METALSOLID, // Твердий метал A3DMTL_METALBARREL, // Металеві бочки A3DMTL_GLASS, // Скло A3DMTL_RUBBER, // Гума A3DMTL_FLESH, // М'язи A3DMTL_WATER, // Вода A3DMTL_EARTH, // Земля };

Процес трасування колізії

// Коли гравець або об'єкт рухається, виконується: 1. A3DModel::RayTrace() або A3DModel::AABBTrace() ↓ 2. Для кожного дочірнього меша (A3DMesh): - Перевірка AABB меша з променем/об'ємом - Якщо перетин → тестування кожної трикутника - Використання A3DMESHPROP для визначення поведінки ↓ 3. Повернення результату: - Точка зіткнення - Нормаль поверхні - Коефіцієнт зіткнення (0-1) - Тип матеріалу

  • Посилання на звуки або GFX ефекти

Висновок

CollisionFile.bmd це MOX файл, який містить:

  • Візуальну геометрію (меші для рендерингу)
  • Колізійну геометрію (ті ж меші використовуються для колізії)
  • Властивості матеріалів (для звукових ефектів та фізики)
  • Текстури (зовнішні посилання)

Колізія реалізована через тестування променів та об'ємів з трикутниками мешів з урахуванням властивостей кожного меша (двосторонність, пропускання куль тощо).

Аналіз Blender експортера

Що вміє експортер

  • Експорт геометрії (меші з вершинами та індексами)
  • Запис UV-координат
  • Запис нормалей
  • Запис кольорів вершин (diffuse, day, night)
  • Запис AABB (вирівняна обмежувальна коробка)
  • Запис базових матеріалів (ambient, diffuse, emissive, specular, power, two_sided)

Що НЕ вміє експортер (проблеми з колізією)

Критична проблема: Експортер НЕ записує властивості колізії (A3DMESHPROP), які визначають поведінку колізії.

Експортер пропускає наступні важливі дані:

Висновок по експортеру

Експортер НЕ повністю підтримує колізію.

Він експортує геометрію (яка використовується для колізійного тестування), але НЕ записує:

  • Прапорці властивостей колізії (A3DMESHPROP.dwProps)
  • Тип матеріалу для звукових ефектів (A3DMATERIALTYPE)

Це означає, що після експорту через цей плагін колізія працюватиме з замовчуванням (всі об'єкти матимуть стандартну поведінку).
